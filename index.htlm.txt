function switchRoom(i) {
  currentRoom = i;
  document.querySelectorAll(".room").forEach((r, j) => r.classList.toggle("active", j === i));
  document.getElementById("chatHead").textContent = "💬 " + ROOMS[i].name;
  renderRoom();
}
function renderRoom() {
  const body = document.getElementById("chatBody");
  body.innerHTML = "";
  ROOMS[currentRoom].seed.forEach(([w, t]) => body.appendChild(makeMsg(w, t, w === "You")));
}
function makeMsg(who, text, me) {
  const m = document.createElement("div");
  m.className = "msg " + (me ? "me" : "them");
  m.innerHTML = `<div class="who">${me ? "You" : who}</div>${text}`;
  return m;
}
function sendMsg() {
  const inp = document.getElementById("chatMsg");
  const t = inp.value.trim();
  if (!t) return;
  const body = document.getElementById("chatBody");
  body.appendChild(makeMsg("You", t, true));
  inp.value = "";
  body.scrollTop = body.scrollHeight;
  setTimeout(() => {
    const replies = ["nice!", "🔥🔥", "lol", "facts", "agreed ✅"];
    body.appendChild(makeMsg("Alex", replies[Math.floor(Math.random() * replies.length)], false));
    body.scrollTop = body.scrollHeight;
  }, 900 + Math.random() * 900);
}

function startDownload() {
  const url = document.getElementById("dlUrl").value.trim();
  const status = document.getElementById("dlStatus");
  if (!url) { status.innerHTML = "⚠️ Please paste a valid link first."; return; }
  fireMonetizedClick();
  const q = document.querySelector(".quality.sel").dataset.q;
  const prog = document.getElementById("dlProgress");
  const bar = document.getElementById("dlBar");
  prog.style.display = "block";
  bar.style.width = "0%";
  status.className = "dl-status";
  status.textContent = "Fetching " + q + "…";
  let p = 0;
  const timer = setInterval(() => {
    p += Math.random() * 9 + 3;
    if (p >= 100) {
      p = 100;
      clearInterval(timer);
      status.innerHTML = `<span class="dl-done">✅ Done! Your ${q} file is ready.</span>`;
    } else {
      status.textContent = `Downloading ${q}… ${Math.floor(p)}%`;
    }
    bar.style.width = p + "%";
  }, 180);
}
